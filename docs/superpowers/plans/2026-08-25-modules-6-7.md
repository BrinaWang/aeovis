```python
# aeo_eval/website_accessibility/robots_checker.py
"""Evaluate robots.txt rules for configured crawlers."""

import urllib.robotparser
import logging
from typing import Dict, Optional

logger = logging.getLogger(__name__)

DEFAULT_CRAWLERS = [
    "OAI-SearchBot",
    "GPTBot",
    "PerplexityBot",
    "Claude-SearchBot",
    "Googlebot",
    "Bingbot",
]

class RobotsChecker:
    """Check whether a crawler is allowed by robots.txt."""
    
    def __init__(self, timeout_seconds: int = 15):
        self.timeout_seconds = timeout_seconds
        self.cache: Dict[str, urllib.robotparser.RobotFileParser] = {}
    
    def is_allowed(self, url: str, crawler: str) -> bool:
        """
        Check if a crawler is allowed to access a URL per robots.txt.
        
        Args:
            url: Full URL to check (e.g., https://www.striim.com/product/oracle-cdc/)
            crawler: Crawler user-agent (e.g., "OAI-SearchBot")
        
        Returns:
            True if allowed, False if disallowed, True if robots.txt missing (assume allowed)
        """
        try:
            # Extract domain from URL
            from urllib.parse import urlparse
            parsed = urlparse(url)
            domain = f"{parsed.scheme}://{parsed.netloc}"
            robots_url = f"{domain}/robots.txt"
            
            # Get or create parser for this domain
            if domain not in self.cache:
                parser = urllib.robotparser.RobotFileParser()
                parser.set_url(robots_url)
                try:
                    parser.read()
                except Exception as e:
                    logger.warning(f"Failed to fetch {robots_url}: {e}")
                    # If robots.txt missing, assume allowed
                    parser = None
                self.cache[domain] = parser
            
            parser = self.cache[domain]
            if parser is None:
                # robots.txt missing/unreachable, assume allowed
                return True
            
            # Check if crawler is allowed
            return parser.can_fetch(crawler, url)
        
        except Exception as e:
            logger.error(f"Error checking robots.txt for {url} / {crawler}: {e}")
            # Fail open: if we can't determine, assume allowed
            return True
```

```python
# aeo_eval/website_accessibility/http_checker.py
"""Check HTTP status, response time, and metadata (canonical, noindex, sitemap)."""

import httpx
import time
import logging
from typing import Dict, Optional, List
from bs4 import BeautifulSoup
from urllib.parse import urljoin

logger = logging.getLogger(__name__)

USER_AGENT = "striim-aeo-monitor/1.0 (internal AEO research)"

class HttpChecker:
    """Check HTTP metadata for a Striim page."""
    
    def __init__(self, timeout_seconds: int = 15, rate_limit_rps: float = 0.5):
        self.timeout_seconds = timeout_seconds
        self.rate_limit_rps = rate_limit_rps
        self.last_request_time: Dict[str, float] = {}
    
    def _rate_limit(self, domain: str):
        """Apply rate limiting per domain."""
        from urllib.parse import urlparse
        parsed = urlparse(domain) if domain.startswith("http") else urlparse(f"https://{domain}")
        domain_key = f"{parsed.scheme}://{parsed.netloc}"
        
        if domain_key in self.last_request_time:
            elapsed = time.time() - self.last_request_time[domain_key]
            wait_time = (1.0 / self.rate_limit_rps) - elapsed
            if wait_time > 0:
                time.sleep(wait_time)
        
        self.last_request_time[domain_key] = time.time()
    
    def check_page(self, url: str) -> Dict:
        """
        Check HTTP status and metadata for a page.
        
        Args:
            url: Full URL to check
        
        Returns:
            Dict with keys:
            - status_code: HTTP status
            - response_time_ms: Round-trip time
            - noindex: bool (True if page has noindex meta tag or header)
            - canonical_url: str or None
            - content_length: bytes
            - fetch_error: str or None (if request failed)
        """
        result = {
            "url": url,
            "status_code": None,
            "response_time_ms": None,
            "noindex": False,
            "canonical_url": None,
            "content_length": 0,
            "fetch_error": None,
        }
        
        try:
            self._rate_limit(url)
            
            start = time.time()
            response = httpx.get(
                url,
                headers={"User-Agent": USER_AGENT},
                follow_redirects=True,
                timeout=self.timeout_seconds,
            )
            elapsed_ms = (time.time() - start) * 1000
            
            result["status_code"] = response.status_code
            result["response_time_ms"] = elapsed_ms
            result["content_length"] = len(response.content)
            
            # Parse HTML for metadata
            if response.status_code == 200 and "text/html" in response.headers.get("content-type", ""):
                soup = BeautifulSoup(response.text, "html.parser")
                
                # Check for noindex
                noindex_meta = soup.find("meta", attrs={"name": "robots", "content": lambda x: x and "noindex" in x})
                if noindex_meta or "noindex" in response.headers.get("x-robots-tag", ""):
                    result["noindex"] = True
                
                # Get canonical URL
                canonical = soup.find("link", attrs={"rel": "canonical"})
                if canonical and canonical.get("href"):
                    result["canonical_url"] = canonical["href"]
                elif "content-location" in response.headers:
                    result["canonical_url"] = response.headers["content-location"]
        
        except Exception as e:
            result["fetch_error"] = str(e)
            logger.error(f"Error fetching {url}: {e}")
        
        return result
```

```python
# aeo_eval/website_accessibility/extractability.py
"""Check content extractability via Trafilatura."""

import httpx
import trafilatura
import logging
from typing import Dict, Optional
import time

logger = logging.getLogger(__name__)

USER_AGENT = "striim-aeo-monitor/1.0 (internal AEO research)"

class ExtractabilityChecker:
    """Check whether page content is extractable without JavaScript."""
    
    def __init__(self, timeout_seconds: int = 15, rate_limit_rps: float = 0.5):
        self.timeout_seconds = timeout_seconds
        self.rate_limit_rps = rate_limit_rps
        self.last_request_time: Dict[str, float] = {}
    
    def _rate_limit(self, domain: str):
        """Apply rate limiting per domain."""
        from urllib.parse import urlparse
        parsed = urlparse(domain) if domain.startswith("http") else urlparse(f"https://{domain}")
        domain_key = f"{parsed.scheme}://{parsed.netloc}"
        
        if domain_key in self.last_request_time:
            elapsed = time.time() - self.last_request_time[domain_key]
            wait_time = (1.0 / self.rate_limit_rps) - elapsed
            if wait_time > 0:
                time.sleep(wait_time)
        
        self.last_request_time[domain_key] = time.time()
    
    def check_extractability(self, url: str) -> Dict:
        """
        Check content extractability for a page.
        
        Args:
            url: Full URL to check
        
        Returns:
            Dict with keys:
            - url: The URL checked
            - extraction_ratio: extracted_chars / raw_html_bytes (0.0-1.0)
            - word_count: Approximate word count of extracted text
            - requires_js_render: bool (True if static extraction too low)
            - result: "well_extractable", "poorly_extractable", "fetch_failed"
            - fetch_error: str or None
        """
        result = {
            "url": url,
            "extraction_ratio": 0.0,
            "word_count": 0,
            "requires_js_render": False,
            "result": "fetch_failed",
            "raw_html_bytes": 0,
            "extracted_text_chars": 0,
            "fetch_error": None,
        }
        
        try:
            self._rate_limit(url)
            
            # Fetch page
            response = httpx.get(
                url,
                headers={"User-Agent": USER_AGENT},
                follow_redirects=True,
                timeout=self.timeout_seconds,
            )
            
            if response.status_code != 200:
                result["fetch_error"] = f"HTTP {response.status_code}"
                return result
            
            raw_html = response.content
            result["raw_html_bytes"] = len(raw_html)
            
            # Extract with Trafilatura
            extracted = trafilatura.extract(raw_html, include_comments=False)
            
            if not extracted:
                result["result"] = "poorly_extractable"
                return result
            
            result["extracted_text_chars"] = len(extracted)
            result["word_count"] = len(extracted.split())
            
            # Calculate ratio
            if result["raw_html_bytes"] > 0:
                result["extraction_ratio"] = result["extracted_text_chars"] / result["raw_html_bytes"]
            
            # Classify based on thresholds
            if result["word_count"] < 300:
                result["result"] = "poorly_extractable"
            elif result["extraction_ratio"] < 0.05:  # 5% = crude threshold
                result["result"] = "poorly_extractable"
            else:
                result["result"] = "well_extractable"
        
        except Exception as e:
            result["fetch_error"] = str(e)
            result["result"] = "fetch_failed"
            logger.error(f"Error checking extractability for {url}: {e}")
        
        return result
```

```python
# aeo_eval/website_accessibility/checker.py
"""Orchestrator for website accessibility checks."""

import logging
from typing import List, Dict
from datetime import datetime
import uuid

from .robots_checker import RobotsChecker
from .http_checker import HttpChecker
from .extractability import ExtractabilityChecker

logger = logging.getLogger(__name__)

DEFAULT_CRAWLERS = [
    "OAI-SearchBot",
    "GPTBot",
    "PerplexityBot",
    "Claude-SearchBot",
    "Googlebot",
    "Bingbot",
]

class WebsiteAccessibilityChecker:
    """Orchestrate all website accessibility checks."""
    
    def __init__(self):
        self.robots_checker = RobotsChecker()
        self.http_checker = HttpChecker()
        self.extractability_checker = ExtractabilityChecker()
    
    def check_pages(
        self,
        important_pages: List[str],
        crawlers: List[str] = None,
    ) -> List[Dict]:
        """
        Check all important pages against all crawlers.
        
        Args:
            important_pages: List of URLs to check
            crawlers: List of crawler user-agents (default: all configured crawlers)
        
        Returns:
            List of result dicts ready for database insertion
        """
        if crawlers is None:
            crawlers = DEFAULT_CRAWLERS
        
        results = []
        
        for url in important_pages:
            logger.info(f"Checking {url}")
            
            # HTTP check (once per URL)
            http_result = self.http_checker.check_page(url)
            
            # Extractability check (once per URL)
            extract_result = self.extractability_checker.check_extractability(url)
            
            # Robots.txt check per crawler
            for crawler in crawlers:
                robots_allowed = self.robots_checker.is_allowed(url, crawler)
                
                # Determine overall result
                overall_result = self._classify_result(
                    http_result,
                    robots_allowed,
                    extract_result,
                )
                
                record = {
                    "id": str(uuid.uuid4()),
                    "striim_url": url,
                    "crawler": crawler,
                    "robots_allowed": 1 if robots_allowed else 0,
                    "in_sitemap": None,  # TODO: implement sitemap check
                    "http_status": http_result.get("status_code"),
                    "response_time_ms": http_result.get("response_time_ms"),
                    "noindex": 1 if http_result.get("noindex") else 0,
                    "canonical_url": http_result.get("canonical_url"),
                    "result": overall_result,
                    "check_timestamp": datetime.utcnow().isoformat(),
                }
                results.append(record)
        
        logger.info(f"Completed checks: {len(results)} records")
        return results
    
    def _classify_result(self, http_result: Dict, robots_allowed: bool, extract_result: Dict) -> str:
        """
        Classify overall accessibility result.
        
        Returns: "publicly_accessible", "blocked_by_robots", "http_error_4xx", "http_error_5xx",
                 "poorly_extractable", "js_dependent", "fetch_failed"
        """
        # HTTP errors take priority
        status = http_result.get("status_code")
        if status and 400 <= status < 500:
            return "http_error_4xx"
        if status and status >= 500:
            return "http_error_5xx"
        
        # Robots.txt block
        if not robots_allowed:
            return "blocked_by_robots"
        
        # Fetch failure
        if http_result.get("fetch_error"):
            return "fetch_failed"
        
        # Extractability issues
        if extract_result.get("result") == "poorly_extractable":
            return "poorly_extractable"
        
        # Default: accessible
        return "publicly_accessible"
```

```python
def store_website_checks(self, records: List[Dict]) -> int:

    Args:
        records: List of dicts from WebsiteAccessibilityChecker.check_pages()

            INSERT INTO website_checks
            (id, striim_url, crawler, robots_allowed, in_sitemap, http_status,
             response_time_ms, noindex, canonical_url, result, check_timestamp)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            record["id"],
            record["striim_url"],
            record["crawler"],
            record["robots_allowed"],

            record["http_status"],
            record["response_time_ms"],
            record["noindex"],
            record["canonical_url"],
            record["result"],
            record["check_timestamp"],
        ))
        inserted += 1
    
    conn.commit()
    conn.close()
    return inserted
```

```python
# aeo_eval/request_logs/parser.py
"""Parse and normalize request logs."""

import json
import logging
from typing import Dict, List, Optional
from datetime import datetime, timedelta
from pathlib import Path
import uuid

logger = logging.getLogger(__name__)

REQUIRED_FIELDS = ["timestamp", "host", "path", "status_code", "user_agent"]
OPTIONAL_FIELDS = ["response_time_ms", "referrer"]

class RequestLogParser:
    """Parse and normalize request-log records."""
    
    def parse_json_line(self, line: str) -> Optional[Dict]:
        """
        Parse a single JSONL record.
        
        Args:
            line: JSON line from log file
        
        Returns:
            Normalized dict or None if parsing fails
        """
        try:
            record = json.loads(line)
            
            # Validate required fields
            for field in REQUIRED_FIELDS:
                if field not in record:
                    logger.warning(f"Missing required field '{field}': {line[:100]}")
                    return None
            
            # Normalize
            normalized = {
                "id": str(uuid.uuid4()),
                "timestamp": record["timestamp"],
                "host": record["host"],
                "path": record["path"],
                "status_code": int(record["status_code"]),
                "user_agent": record["user_agent"],
                "response_time_ms": record.get("response_time_ms"),
                "referrer": record.get("referrer"),
                # Strip query params and IP addresses
                "normalized_path": self._normalize_path(record["path"]),
            }
            
            return normalized
        
        except json.JSONDecodeError as e:
            logger.error(f"JSON parse error: {e}")
            return None
        except Exception as e:
            logger.error(f"Error parsing log line: {e}")
            return None
    
    def _normalize_path(self, path: str) -> str:
        """Remove query parameters from path."""
        if "?" in path:
            return path.split("?")[0]
        return path
    
    def parse_file(self, file_path: str, days_back: int = 90) -> List[Dict]:
        """
        Parse a JSONL or JSON array log file.
        
        Args:
            file_path: Path to log file
            days_back: Only include records from last N days
        
        Returns:
            List of normalized records
        """
        records = []
        cutoff = datetime.utcnow() - timedelta(days=days_back)
        
        path = Path(file_path)
        if not path.exists():
            logger.error(f"Log file not found: {file_path}")
            return records
        
        try:
            with open(path, "r") as f:
                for line_num, line in enumerate(f, 1):
                    line = line.strip()
                    if not line:
                        continue
                    
                    record = self.parse_json_line(line)
                    if not record:
                        continue
                    
                    # Filter by date
                    try:
                        record_time = datetime.fromisoformat(record["timestamp"].replace("Z", "+00:00"))
                        if record_time < cutoff:
                            continue
                    except Exception as e:
                        logger.warning(f"Could not parse timestamp on line {line_num}: {e}")
                        continue
                    
                    records.append(record)
            
            logger.info(f"Parsed {len(records)} records from {file_path}")
            return records
        
        except Exception as e:
            logger.error(f"Error reading log file {file_path}: {e}")
            return records
```

```python
# aeo_eval/request_logs/classifier.py
"""Classify crawlers and user-agents from request logs."""

import re
import logging
from typing import Dict

logger = logging.getLogger(__name__)

# Known AI crawlers
KNOWN_AI_CRAWLERS = {
    "OAI-SearchBot": "OpenAI Search",
    "GPTBot": "OpenAI GPT",
    "PerplexityBot": "Perplexity",
    "Claude-SearchBot": "Anthropic Claude",
    "ClaudeBot": "Anthropic Claude",
}

# Delegated agents (used by LLMs to fetch on behalf of a user)
DELEGATED_AGENT_PATTERNS = {
    r"ChatGPT-User": "ChatGPT (delegated)",
    r"Perplexity-User": "Perplexity (delegated)",
    r"agent-(\w+)": "agent-* (delegated)",
}

# Search crawlers
SEARCH_CRAWLERS = {
    "Googlebot": "Google",
    "Bingbot": "Bing",
    "Slurp": "Yahoo",
    "DuckDuckGo-Bot": "DuckDuckGo",
}

class CrawlerClassifier:
    """Classify user-agents into categories."""
    
    def classify(self, user_agent: str) -> Dict:
        """
        Classify a user-agent string.
        
        Args:
            user_agent: User-agent header value
        
        Returns:
            Dict with keys: class, matched_pattern, tool_name, model_hint, known_crawler
            - class: "known_ai_crawler" | "delegated_agent" | "search_crawler" | "human_browser" | "unknown"
            - known_crawler: bool
        """
        if not user_agent:
            return {
                "class": "unknown",
                "matched_pattern": None,
                "tool_name": None,
                "model_hint": None,
                "known_crawler": False,
            }
        
        user_agent_lower = user_agent.lower()
        
        # Check known AI crawlers
        for crawler, name in KNOWN_AI_CRAWLERS.items():
            if crawler.lower() in user_agent_lower:
                return {
                    "class": "known_ai_crawler",
                    "matched_pattern": crawler,
                    "tool_name": name,
                    "model_hint": self._extract_model_hint(user_agent),
                    "known_crawler": True,
                }
        
        # Check delegated agents
        for pattern, name in DELEGATED_AGENT_PATTERNS.items():
            if re.search(pattern, user_agent, re.IGNORECASE):
                match = re.search(pattern, user_agent, re.IGNORECASE)
                tool_name = match.group(1) if match.groups() else name
                return {
                    "class": "delegated_agent",
                    "matched_pattern": pattern,
                    "tool_name": tool_name,
                    "model_hint": self._extract_model_hint(user_agent),
                    "known_crawler": False,
                }
        
        # Check search crawlers
        for crawler, name in SEARCH_CRAWLERS.items():
            if crawler.lower() in user_agent_lower:
                return {
                    "class": "search_crawler",
                    "matched_pattern": crawler,
                    "tool_name": name,
                    "model_hint": None,
                    "known_crawler": True,
                }
        
        # Browser heuristics
        if any(b.lower() in user_agent_lower for b in ["chrome", "firefox", "safari", "edge", "mozilla"]):
            return {
                "class": "human_browser",
                "matched_pattern": "browser_heuristic",
                "tool_name": None,
                "model_hint": None,
                "known_crawler": False,
            }
        
        # Default: unknown
        return {
            "class": "unknown",
            "matched_pattern": None,
            "tool_name": None,
            "model_hint": self._extract_model_hint(user_agent),
            "known_crawler": False,
        }
    
    def _extract_model_hint(self, user_agent: str) -> str:
        """Try to extract model name from user-agent."""
        # Look for patterns like (claude-sonnet-4) or claude-sonnet-4-6
        match = re.search(r"\(?(claude-[a-z0-9\-]+)\)?", user_agent, re.IGNORECASE)
        if match:
            return match.group(1)
        
        match = re.search(r"\(?(gpt-[a-z0-9\-]+)\)?", user_agent, re.IGNORECASE)
        if match:
            return match.group(1)
        
        return None
```

```python
# aeo_eval/request_logs/analyzer.py
"""Analyze request logs for crawler activity patterns."""

import logging
from typing import Dict, List
from collections import defaultdict
from datetime import datetime

from .classifier import CrawlerClassifier

logger = logging.getLogger(__name__)

class RequestLogAnalyzer:
    """Analyze parsed request logs."""
    
    def __init__(self):
        self.classifier = CrawlerClassifier()
    
    def analyze(self, records: List[Dict]) -> List[Dict]:
        """
        Normalize and analyze request log records.
        
        Args:
            records: List of parsed records from RequestLogParser
        
        Returns:
            List of crawler_log records ready for database insertion
        """
        normalized = []
        
        for record in records:
            # Classify the crawler
            classification = self.classifier.classify(record["user_agent"])
            
            # Normalize to crawler_logs table schema
            log_record = {
                "id": record["id"],
                "timestamp": record["timestamp"],
                "host": record["host"],
                "path": record["normalized_path"],
                "crawler": classification.get("matched_pattern") or record["user_agent"][:50],
                "http_status": record["status_code"],
                "response_time_ms": record.get("response_time_ms"),
                "edge_action": self._classify_edge_action(record["status_code"]),
                "log_source": "uploaded",  # Will be set to file path if from file
                "ua_classification": {
                    "class": classification["class"],
                    "matched_pattern": classification["matched_pattern"],
                    "tool_name": classification["tool_name"],
                    "model_hint": classification["model_hint"],
                    "known_crawler": classification["known_crawler"],
                },
            }
            
            normalized.append(log_record)
        
        logger.info(f"Analyzed {len(normalized)} log records")
        return normalized
    
    def _classify_edge_action(self, status_code: int) -> str:
        """Classify edge action based on HTTP status."""
        if 200 <= status_code < 300:
            return "allowed"
        elif status_code == 403:
            return "blocked"
        elif status_code == 429:
            return "rate_limited"
        else:
            return "error"
    
    def summarize_activity(self, records: List[Dict]) -> Dict:
        """
        Summarize crawler activity patterns.
        
        Returns:
            Dict with keys: total_records, by_crawler, by_status, failures
        """
        summary = {
            "total_records": len(records),
            "by_crawler": defaultdict(int),
            "by_status": defaultdict(int),
            "failures": [],
        }
        
        for record in records:
            crawler = record.get("crawler", "unknown")
            status = record.get("http_status", 0)
            
            summary["by_crawler"][crawler] += 1
            summary["by_status"][status] += 1
            
            if status >= 400:
                summary["failures"].append({
                    "crawler": crawler,
                    "path": record.get("path"),
                    "status": status,
                    "timestamp": record.get("timestamp"),
                })
        
        return summary
```
